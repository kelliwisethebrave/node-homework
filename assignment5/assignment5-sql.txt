#1
#Find the total price of each of the first 5 orders, ordered by order_id.
#returning two columns order_id and total_price

SELECT orders.order_id, sum(products.price * line_items.quantity) AS total_price FROM orders JOIN line_items ON orders.order_id = line_items.order_id JOIN products ON line_items.product_id = products.product_id GROUP BY orders.order_id ORDER BY orders.order_id LIMIT 5;

#2
#find the average price of each customer's orders
#return the results ordered by customer name
#FIRST the total price of each order which is the subquery

SELECT cust.customer_name, AVG(order_totals.total_price) AS average_order_price FROM customers cust JOIN (SELECT orders.customer_id, orders.order_id, sum(products.price * line_items.quantity) AS total_price FROM orders JOIN line_items ON orders.order_id = line_items.order_id JOIN products ON line_items.product_id = products.product_id GROUP BY orders.customer_id, orders.order_id) AS order_totals ON cust.customer_id = order_totals.customer_id GROUP BY cust.customer_id, cust.customer_name ORDER BY cust.customer_name;

# find all employees who have more than 5 order_totals
# return first_name, last_name, and the count of orders as order_count
# sort the results by last_name (ORDER BY)

SELECT employees.first_name, employees.last_name, COUNT(orders.order_id) AS order_count FROM employees JOIN orders ON employees.employee_id = orders.employee_id GROUP BY employees.employee_id, first_name, last_name HAVING COUNT(orders.order_id) > 5 ORDER BY employees.last_name;
